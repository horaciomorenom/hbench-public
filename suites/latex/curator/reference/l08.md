```latex
\documentclass[11pt]{article}
\usepackage{amsmath}
\usepackage{algorithm}
\usepackage{algpseudocode}
\begin{document}
\begin{algorithm}[ht]
\caption{Preconditioned conjugate gradients}
\label{alg:pcg}
\begin{algorithmic}[1]
\Require matrix $A$, right-hand side $b$, preconditioner $M$, initial guess $x_0$, tolerance $\varepsilon$
\Ensure approximate solution $x$
\State $r_0 \gets b - A x_0$
\State $z_0 \gets M^{-1} r_0$
\State $p_0 \gets z_0$
\State $k \gets 0$
\While{$\|r_k\| > \varepsilon$} \Comment{stop at the tolerance}
  \State $\alpha_k \gets \frac{r_k^T z_k}{p_k^T A p_k}$
  \State $x_{k+1} \gets x_k + \alpha_k p_k$
  \State $r_{k+1} \gets r_k - \alpha_k A p_k$
  \State $z_{k+1} \gets M^{-1} r_{k+1}$
  \State $\beta_k \gets \frac{r_{k+1}^T z_{k+1}}{r_k^T z_k}$
  \State $p_{k+1} \gets z_{k+1} + \beta_k p_k$
  \State $k \gets k + 1$
\EndWhile
\State \Return $x_k$
\end{algorithmic}
\end{algorithm}

Algorithm~\ref{alg:pcg} needs one preconditioner solve per iteration.
\end{document}
```
