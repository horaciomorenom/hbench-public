omega = -v* ky / (1 + k^2), from q = -(1+k^2) phi.
```python
import numpy as np
def omega(kx, ky, v_star=1.0):
    return -v_star * ky / (1 + kx**2 + ky**2)
def solve_hm(N=32, dt=0.1, T=100.0):
    L = 20*np.pi; vs = 1.0
    g = np.arange(N) * L / N; X, Y = np.meshgrid(g, g, indexing="ij")
    k1 = 2*np.pi*np.fft.fftfreq(N, d=L/N); KX, KY = np.meshgrid(k1, k1, indexing="ij"); K2 = KX**2 + KY**2
    kx, ky = 2*np.pi*3/L, 2*np.pi*2/L
    ph = np.fft.fft2(np.cos(kx*X + ky*Y))
    f = lambda p: (vs * 1j * KY / (1 + K2)) * p
    for _ in range(int(round(T/dt))):
        a = f(ph); b = f(ph + .5*dt*a); c = f(ph + .5*dt*b); d = f(ph + dt*c)
        ph = ph + dt/6*(a + 2*b + 2*c + d)
    return np.fft.ifft2(ph).real
```
