import numpy as np, matplotlib.pyplot as plt
def plot_convergence(h, err, title="Convergence"):
    h, err = np.asarray(h, float), np.asarray(err, float)
    p, c = np.polyfit(np.log(h), np.log(err), 1)
    fig, ax = plt.subplots(figsize=(4, 3.2))
    ax.loglog(h, err, "o-", label="error")
    ax.loglog(h, np.exp(c) * h**p, "--", label=f"fit, order {p:.2f}")
    ax.text(0.05, 0.9, f"observed order = {p:.2f}", transform=ax.transAxes)
    ax.set_xlabel("h"); ax.set_ylabel("error"); ax.legend(); ax.set_title(title)
    fig.tight_layout()
    return p, fig
