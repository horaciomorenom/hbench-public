```python
import numpy as np
def householder_qr(A):
    A = A.astype(float).copy(); m, n = A.shape; Q = np.eye(m)
    for k in range(n):
        x = A[k:, k]; v = x.copy(); v[0] += np.sign(x[0] if x[0] != 0 else 1.0) * np.linalg.norm(x)
        v /= np.linalg.norm(v)
        A[k:, :] -= 2 * np.outer(v, v @ A[k:, :]); Q[:, k:] -= 2 * np.outer(Q[:, k:] @ v, v)
    return Q, np.triu(A)
def lstsq_qr(A, b):
    Q, R = householder_qr(A); n = A.shape[1]
    return np.linalg.solve(R[:n], Q[:, :n].T @ b)
```
Forming A^T A squares the condition number, so the normal equations lose accuracy.
