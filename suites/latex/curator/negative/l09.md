\documentclass[11pt]{article}
\usepackage{tikz}
\usetikzlibrary{positioning,shapes.geometric,arrows.meta}
\begin{document}
\begin{center}
\begin{tikzpicture}[node distance=-5mm and 15mm,
  box/.style={draw, rectangle, minimum width=30mm, minimum height=8mm},
  >=Stealth]
  \node[box] (mesh) {Generate mesh};
  \node[box, below=of mesh] (assemble) {Assemble system};
  \node[box, below=of assemble] (solve) {Solve system};
  \node[draw, diamond, aspect=2, below=of solve] (check) {Error small?};
  \node[box, rounded corners, below=of check] (output) {Write output};
  \node[box, right=of check] (refine) {Refine mesh};
  \draw[->] (mesh) -- (assemble);
  \draw[->] (assemble) -- (solve);
  \draw[->] (solve) -- (check);
  \draw[->] (check) -- node[right] {yes} (output);
  \draw[->] (check) -- node[above] {no} (refine);
  \draw[->] (refine) -- (output);
\end{tikzpicture}
\end{center}
