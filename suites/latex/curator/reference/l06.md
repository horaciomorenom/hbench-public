\begin{document}
\section{Stability}
\begin{definition}\label{def:stable}
A scheme is stable if its amplification factor satisfies $|g| \le 1$ for all wavenumbers.
\end{definition}
\begin{lemma}\label{lem:bound}
The explicit Euler scheme for $y' = \lambda y$ has $g = 1 + \lambda \Delta t$.
\end{lemma}
\begin{theorem}\label{thm:main}
Explicit Euler is stable for real $\lambda < 0$ if and only if $\Delta t \le 2/|\lambda|$.
\end{theorem}
\begin{proof}
By \Cref{lem:bound}, the condition $|1 + \lambda \Delta t| \le 1$ is required by \Cref{def:stable}. Solving this inequality gives the bound.
\end{proof}
This completes the analysis of \Cref{thm:main}.
\end{document}
